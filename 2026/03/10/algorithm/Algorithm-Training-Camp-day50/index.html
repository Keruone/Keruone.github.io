题目答案及分析

99. 计数孤岛

思路介绍：

  • 遍历。
  • 当什么时候，踩到陆地了，那就说明至少在一个岛屿上。
  • 利用 dfs 或 bfs ，将整个岛屿探索完毕
  • 每个探索过的地方都打上标记
  • 遍历 到 有标记的地点，就跳过（节省遍历时间）

而对于 dfs 和 bfs 的具体实现，就递归和队列。不展开细说了。

100. 最大岛屿的面积

跟上一题思路一致，不过探索的同时还要计数。最后比较每一块的土地即可

dfs

bfs

#include<iostream>
#include<vector>
#include<queue>
using namespace std;

int dir[4][2] = {{0, -1}, {0, 1}, {-1, 0}, {1, 0}}; // up down left right
int bfs(vector<vector<int>>& map, vector<vector<bool>>& visited, int start_y, int start_x){
    // 输入的 y, x 一定是有效的
    queue<pair<int, int>> que;
    que.push({start_y, start_x});
    int island_area = 1;    // 起始面积
    while(!que.empty()){
        int temp_y = que.front().first;
        int temp_x = que.front().second;
        que.pop();
        for(int i = 0; i < 4; i++){
            int curr_y = temp_y + dir[i][0];
            int curr_x = temp_x + dir[i][1];
            if( curr_y < 0 || curr_y >= map.size() ||\
                curr_x < 0 || curr_x >= map[0].size()
            ) continue;
            if(map[curr_y][curr_x] == 1 && !visited[curr_y][curr_x]){
                visited[curr_y][curr_x] = true;
                que.push(pair<int, int>(curr_y, curr_x));
                island_area++;
            }
        }
    }
    return island_area;
}
int main(void){
    int M, N;
    cin >> N >> M;
    vector<vector<int>> map(N, vector<int>(M, 0));
    vector<vector<bool>> visited(N, vector<bool>(M, false));
    for(int i = 0; i < N; i++){
        for(int j = 0; j < M; j++){
            cin >> map[i][j];
        }
    }
    int island_area = 0;
    int island_area_max = 0;
    for(int i = 0; i < N; i++){
        for(int j = 0; j < M; j++){
            if(!visited[i][j] && map[i][j] == 1){
                visited[i][j] = true;
                island_area = bfs(map, visited, i, j);
                island_area_max = max(island_area, island_area_max);
            }
        }
    }
    
    cout << island_area_max;
    return 0;
}
